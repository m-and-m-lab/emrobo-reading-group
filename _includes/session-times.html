-%}{%- if st_h12 == 0 -%}{%- assign st_h12 = 12 -%}{%- endif -%} {%- assign st_eh12 = st_eh | modulo: 12 -%}{%- if st_eh12 == 0 -%}{%- assign st_eh12 = 12 -%}{%- endif -%} {%- if st_h < 12 -%}{%-
